49 /// @brief Compile and link shader program from sources.
50 Result<void> compile(QOpenGLFunctions_4_1_Core* gl,
51 const char* vertexSource,
52 const char* fragmentSource) {
53 cleanup();
54 m_gl = gl;
55
56 GLuint vertexShader = gl->glCreateShader(GL_VERTEX_SHADER);
57 gl->glShaderSource(vertexShader, 1, &vertexSource, nullptr);
58 gl->glCompileShader(vertexShader);
59
60 GLint success = 0;
61 gl->glGetShaderiv(vertexShader, GL_COMPILE_STATUS, &success);
62 if (!success) {
63 char log[512];
64 gl->glGetShaderInfoLog(vertexShader, 512, nullptr, log);
65 gl->glDeleteShader(vertexShader);
66 return std::unexpected(Error::shaderError(QString("Vertex shader: %1").arg(log)));
67 }
68
69 GLuint fragmentShader = gl->glCreateShader(GL_FRAGMENT_SHADER);
70 gl->glShaderSource(fragmentShader, 1, &fragmentSource, nullptr);
71 gl->glCompileShader(fragmentShader);
72
73 gl->glGetShaderiv(fragmentShader, GL_COMPILE_STATUS, &success);
74 if (!success) {
75 char log[512];
76 gl->glGetShaderInfoLog(fragmentShader, 512, nullptr, log);
77 gl->glDeleteShader(vertexShader);
78 gl->glDeleteShader(fragmentShader);
79 return std::unexpected(Error::shaderError(QString("Fragment shader: %1").arg(log)));
80 }
81
82 m_program = gl->glCreateProgram();
83 gl->glAttachShader(m_program, vertexShader);
84 gl->glAttachShader(m_program, fragmentShader);
85 gl->glLinkProgram(m_program);
86
87 gl->glGetProgramiv(m_program, GL_LINK_STATUS, &success);
88 if (!success) {
89 char log[512];
90 gl->glGetProgramInfoLog(m_program, 512, nullptr, log);
91 gl->glDeleteShader(vertexShader);
92 gl->glDeleteShader(fragmentShader);
93 gl->glDeleteProgram(m_program);
94 m_program = 0;
95 return std::unexpected(Error::shaderError(QString("Program link: %1").arg(log)));
96 }
97
98 gl->glDeleteShader(vertexShader);
99 gl->glDeleteShader(fragmentShader);
100
101 return {};
102 }
