53/// @brief Axis-aligned bounding box for spatial queries.
54struct AABB {
55 Vec3 min{std::numeric_limits<float>::max(),
56 std::numeric_limits<float>::max(),
57 std::numeric_limits<float>::max()};
58 Vec3 max{std::numeric_limits<float>::lowest(),
59 std::numeric_limits<float>::lowest(),
60 std::numeric_limits<float>::lowest()};
61
62 [[nodiscard]] Vec3 center() const noexcept { return (min + max) * 0.5f; }
63 [[nodiscard]] Vec3 size() const noexcept { return max - min; }
64 [[nodiscard]] bool isValid() const noexcept {
65 return min.x() <= max.x() && min.y() <= max.y() && min.z() <= max.z();
66 }
67
68 void expand(const Vec3& point) noexcept {
69 min.setX(std::min(min.x(), point.x()));
70 min.setY(std::min(min.y(), point.y()));
71 min.setZ(std::min(min.z(), point.z()));
72 max.setX(std::max(max.x(), point.x()));
73 max.setY(std::max(max.y(), point.y()));
74 max.setZ(std::max(max.z(), point.z()));
75 }

102 /// @brief Decomposes a 4x4 matrix into TRS components.
103 /// @param matrix The transformation matrix to decompose.
104 /// @return Transform with extracted position, rotation, and scale.
105 static Transform fromMatrix(const Mat4& matrix) noexcept {
106 Transform t;
107 t.position = Vec3(matrix(0, 3), matrix(1, 3), matrix(2, 3));
108
109 Vec3 col0(matrix(0, 0), matrix(1, 0), matrix(2, 0));
110 Vec3 col1(matrix(0, 1), matrix(1, 1), matrix(2, 1));
111 Vec3 col2(matrix(0, 2), matrix(1, 2), matrix(2, 2));
112 t.scale = Vec3(col0.length(), col1.length(), col2.length());
113
114 if (t.scale.x() > 0.0f) col0 /= t.scale.x();
115 if (t.scale.y() > 0.0f) col1 /= t.scale.y();
116 if (t.scale.z() > 0.0f) col2 /= t.scale.z();
117
118 QMatrix3x3 rotMat;
119 rotMat(0, 0) = col0.x(); rotMat(1, 0) = col0.y(); rotMat(2, 0) = col0.z();
120 rotMat(0, 1) = col1.x(); rotMat(1, 1) = col1.y(); rotMat(2, 1) = col1.z();
121 rotMat(0, 2) = col2.x(); rotMat(1, 2) = col2.y(); rotMat(2, 2) = col2.z();
122 t.rotation = Quat::fromRotationMatrix(rotMat);
123
124 return t;
125 }
126};
