52void CadViewWidget::mousePressEvent(QMouseEvent* event) {
53 m_lastMousePos = event->pos();
54
55 if (event->button() == Qt::LeftButton) {
56 if (event->modifiers() & Qt::AltModifier) {
57 m_orbiting = true;
58 } else {
59 // Object picking
60 handlePicking(event->pos().x(), event->pos().y());
61 }
62 } else if (event->button() == Qt::MiddleButton) {
63 m_panning = true;
64 } else if (event->button() == Qt::RightButton) {
65 if (event->modifiers() & Qt::AltModifier) {
66 m_orbiting = true;
67 } else {
68 emit contextMenuRequested(event->globalPosition().toPoint());
69 }
70 }
71
72 event->accept();
73}

75void CadViewWidget::mouseMoveEvent(QMouseEvent* event) {

77 m_lastMousePos = event->pos();
78
79 if (m_orbiting) {

83 m_camera.pan(static_cast<float>(delta.x()), static_cast<float>(delta.y()));
84 update();

111
112 if (hit) {
