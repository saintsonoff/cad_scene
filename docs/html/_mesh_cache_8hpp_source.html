28 /// @brief Get existing buffer or create new one from geometry data.
29 GlBuffer* getOrCreate(ObjectId id, const GeometryData& data) {
30 auto it = m_cache.find(id);
31 if (it != m_cache.end()) {
32 return &it->second;
33 }
34

36 auto& buffer = m_cache[id];
37 buffer.create(m_gl, data);

51 }

56 buffer.cleanup();
