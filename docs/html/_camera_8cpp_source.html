36Vec3 Camera::position() const {
37 const float yawRad = m_yaw * constants::DegToRad;
38 const float pitchRad = m_pitch * constants::DegToRad;
39
40 const float cosP = std::cos(pitchRad);
41 const float sinP = std::sin(pitchRad);
42 const float cosY = std::cos(yawRad);
43 const float sinY = std::sin(yawRad);
44
45 Vec3 offset(
46 m_distance * cosP * sinY,
47 m_distance * sinP,
48 m_distance * cosP * cosY
49 );
50
51 return m_target + offset;

60void Camera::pan(float deltaX, float deltaY) {
61 // Calculate camera right and up vectors
62 Vec3 forward = (m_target - position()).normalized();
63 Vec3 right = Vec3::crossProduct(forward, Vec3(0.0f, 1.0f, 0.0f)).normalized();
64 Vec3 up = Vec3::crossProduct(right, forward).normalized();
65
66 float panScale = m_distance * constants::PanSensitivity;
67 m_target += right * (-deltaX * panScale);
68 m_target += up * (deltaY * panScale);
69}

84void Camera::frameBox(const AABB& bbox) {
85 if (!bbox.isValid()) {
86 return;
87 }
88
89 m_target = bbox.center();
90
91 // Calculate distance to fit box in view
92 Vec3 size = bbox.size();
93 float maxDim = std::max({size.x(), size.y(), size.z()});
94 float halfFovRad = (m_fov * 0.5f) * constants::DegToRad;

109Ray Camera::screenToRay(int x, int y) const {
110 float ndcX = (2.0f * static_cast<float>(x) / static_cast<float>(m_viewportWidth)) - 1.0f;
111 float ndcY = 1.0f - (2.0f * static_cast<float>(y) / static_cast<float>(m_viewportHeight));
112
113 Vec4 clipNear(ndcX, ndcY, -1.0f, 1.0f);
114 Vec4 clipFar(ndcX, ndcY, 1.0f, 1.0f);
115
116 Mat4 invVP = viewProjectionMatrix().inverted();
117 Vec4 worldNear = invVP.map(clipNear);
118 Vec4 worldFar = invVP.map(clipFar);
119
120 Vec3 near3(worldNear.x() / worldNear.w(),
121 worldNear.y() / worldNear.w(),
122 worldNear.z() / worldNear.w());
123 Vec3 far3(worldFar.x() / worldFar.w(),
124 worldFar.y() / worldFar.w(),
125 worldFar.z() / worldFar.w());
126
127 return Ray{near3, (far3 - near3).normalized()};
128}
