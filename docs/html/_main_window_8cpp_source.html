149 // Add toggle actions to view menu
150 QMenu* viewMenu = nullptr;

152 if (action->text() == tr("&View")) {
153 viewMenu = action->menu();
154 break;
155 }
156 }
157 if (viewMenu) {

159 viewMenu->addAction(m_sideBarDock->toggleViewAction());

163void MainWindow::connectSignals() {
164 // Application signals
165 connect(&m_app, &Application::fileLoaded, this, &MainWindow::onFileLoaded);
166 connect(&m_app, &Application::errorOccurred, this, &MainWindow::onErrorOccurred);
167 connect(&m_app.commandHistory(), &CommandHistory::historyChanged, this, &MainWindow::onHistoryChanged);
168
169 // Sidebar signals
170 connect(m_sideBar, &SideBarWidget::requestUpdate, m_viewport, QOverload<>::of(&QWidget::update));
171 connect(m_sideBar, &SideBarWidget::cameraChanged, this, [this]() {
172 m_sideBar->refreshFromCamera();
173 });
174
175 // Scene tree signals
176 connect(m_sceneTree, &SceneTreeWidget::requestUpdate, m_viewport, QOverload<>::of(&QWidget::update));
177 connect(m_sceneTree, &SceneTreeWidget::focusObjectRequested, this, &MainWindow::onFocusObjectRequested);
178
179 // Viewport signals
180 connect(m_viewport, &CadViewWidget::objectClicked, this, [this](ObjectId id) {
181 Q_UNUSED(id);
182 m_sideBar->refreshFromCamera();
183 });
184}

186void MainWindow::updateUndoRedoActions() {
187 m_undoAction->setEnabled(m_app.commandHistory().canUndo());
188 m_redoAction->setEnabled(m_app.commandHistory().canRedo());
189
190 if (m_app.commandHistory().canUndo()) {
191 m_undoAction->setText(tr("&Undo %1").arg(m_app.commandHistory().undoDescription()));
192 } else {
193 m_undoAction->setText(tr("&Undo"));
194 }
195
196 if (m_app.commandHistory().canRedo()) {
197 m_redoAction->setText(tr("&Redo %1").arg(m_app.commandHistory().redoDescription()));
198 } else {
199 m_redoAction->setText(tr("&Redo"));
200 }
