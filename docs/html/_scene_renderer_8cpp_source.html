13Result<void> SceneRenderer::initialize(QOpenGLFunctions_4_1_Core* gl) {
14 m_gl = gl;
15 m_meshCache.setGlFunctions(gl);
16
17 auto pbrResult = m_pbrShader.compile(gl, shaders::PbrVertexShader, shaders::PbrFragmentShader);
18 if (!pbrResult) {
19 return std::unexpected(pbrResult.error());
20 }
21
22 m_pbrShader.bind();
23 m_locModel = m_pbrShader.uniformLocation("uModel");
24 m_locViewProjection = m_pbrShader.uniformLocation("uViewProjection");
25 m_locNormalMatrix = m_pbrShader.uniformLocation("uNormalMatrix");
26 m_locCameraPos = m_pbrShader.uniformLocation("uCameraPos");
27 m_locBaseColor = m_pbrShader.uniformLocation("uBaseColor");
28 m_locMetallic = m_pbrShader.uniformLocation("uMetallic");
29 m_locRoughness = m_pbrShader.uniformLocation("uRoughness");
30 m_locShininess = m_pbrShader.uniformLocation("uShininess");
31 m_locSelected = m_pbrShader.uniformLocation("uSelected");
32 m_pbrShader.unbind();
33
34 auto gridResult = m_gridShader.compile(gl, shaders::GridVertexShader, shaders::GridFragmentShader);
35 if (!gridResult) {
36 return std::unexpected(gridResult.error());
37 }
38
39 GeometryData gridData;
40 const float gridSize = 50.0f;
41 const int gridLines = 100;
42 const float step = gridSize * 2.0f / static_cast<float>(gridLines);
43
44 for (int i = 0; i <= gridLines; ++i) {
45 float pos = -gridSize + static_cast<float>(i) * step;
46
47 gridData.vertices.push_back({{-gridSize, 0.0f, pos}, {0.0f, 1.0f, 0.0f}, {0.0f, 0.0f}});
48 gridData.vertices.push_back({{ gridSize, 0.0f, pos}, {0.0f, 1.0f, 0.0f}, {1.0f, 0.0f}});
49
50 gridData.vertices.push_back({{pos, 0.0f, -gridSize}, {0.0f, 1.0f, 0.0f}, {0.0f, 0.0f}});
51 gridData.vertices.push_back({{pos, 0.0f, gridSize}, {0.0f, 1.0f, 0.0f}, {0.0f, 1.0f}});
52 }
53
54 for (std::uint32_t i = 0; i < gridData.vertices.size(); ++i) {
55 gridData.indices.push_back(i);
56 }
57
58 m_gridBuffer.create(gl, gridData);
59
60 gl->glEnable(GL_DEPTH_TEST);
61 gl->glEnable(GL_CULL_FACE);
62 gl->glCullFace(GL_BACK);
63 gl->glEnable(GL_BLEND);
64 gl->glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
65
66 return {};
67}

97void SceneRenderer::renderObject(const SceneObject& obj, const Camera& camera) {
98 auto* buffer = m_meshCache.getOrCreate(obj.id(), obj.geometry());
99 if (!buffer || !buffer->isValid()) {
100 return;
101 }
102
103 m_pbrShader.bind();
104
105 // Matrices
106 Mat4 model = obj.modelMatrix();
107 Mat4 viewProj = camera.viewProjectionMatrix();
108
109 // Normal matrix (inverse transpose of upper-left 3x3 of model matrix)
110 QMatrix3x3 normalMat = model.normalMatrix();
111
112 m_pbrShader.setUniform(m_locModel, model);
113 m_pbrShader.setUniform(m_locViewProjection, viewProj);
114
115 if (m_locNormalMatrix >= 0) {
116 m_gl->glUniformMatrix3fv(m_locNormalMatrix, 1, GL_FALSE, normalMat.constData());
117 }
118
119 m_pbrShader.setUniform(m_locCameraPos, camera.position());
120
121 // Material
122 const Material& mat = obj.material();
123 m_pbrShader.setUniform(m_locBaseColor, mat.baseColorVec3());
124 m_pbrShader.setUniform(m_locMetallic, mat.metallic);
125 m_pbrShader.setUniform(m_locRoughness, mat.roughness);
126 m_pbrShader.setUniform(m_locShininess, mat.shininess);
127 m_pbrShader.setUniform(m_locSelected, obj.isSelected() ? 1.0f : 0.0f);
128
129 buffer->draw();
130
131 m_pbrShader.unbind();
132}

134void SceneRenderer::renderGrid(const Camera& camera) {
135 if (!m_gridBuffer.isValid()) {
136 return;
137 }
138
139 m_gridShader.bind();
140
141 GLint locVP = m_gridShader.uniformLocation("uViewProjection");
142 m_gridShader.setUniform(locVP, camera.viewProjectionMatrix());
143
144 m_gl->glDisable(GL_CULL_FACE);
145 m_gl->glDepthMask(GL_FALSE);
146
147 m_gridBuffer.bind();
148 m_gl->glDrawElements(GL_LINES, m_gridBuffer.indexCount(), GL_UNSIGNED_INT, nullptr);
149 m_gridBuffer.unbind();
150
151 m_gl->glDepthMask(GL_TRUE);
152 m_gl->glEnable(GL_CULL_FACE);
153
154 m_gridShader.unbind();
155}
