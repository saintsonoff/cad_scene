31bool Scene::removeObject(ObjectId id) {
32 auto indexIt = m_objectIndex.find(id);
33 if (indexIt == m_objectIndex.end()) {
34 return false;
35 }
36
37 auto it = std::ranges::find_if(m_objects, [id](const auto& obj) {
38 return obj->id() == id;

40
41 if (it != m_objects.end()) {
42 m_objects.erase(it);
43 }
44
45 m_objectIndex.erase(indexIt);
46
47 if (m_selectedId == id) {
48 clearSelection();
49 }
50
51 emit objectRemoved(id);
52 return true;

129std::optional<ObjectId> Scene::raycast(const Ray& ray) const {
130 float closestT = std::numeric_limits<float>::max();
131 ObjectId closestId = InvalidObjectId;
132
133 // Precompute inverse direction to avoid division in inner loop.
134 constexpr float epsilon = 1e-8f;
135 auto safeInverse = [](float val) -> float {
136 return std::abs(val) > epsilon ? 1.0f / val : std::copysign(1e8f, val);
137 };
138
139 const Vec3 invDir(
140 safeInverse(ray.direction.x()),
141 safeInverse(ray.direction.y()),
142 safeInverse(ray.direction.z())
143 );
144
145 for (const auto& obj : m_objects) {
146 if (!obj->isVisible()) {
147 continue;
148 }
149
150 AABB bbox = obj->boundingBox();
151 if (!bbox.isValid()) {
152 continue;
153 }
154
155 // Ray-AABB intersection test using precomputed inverse direction
156 float tmin = (bbox.min.x() - ray.origin.x()) * invDir.x();
157 float tmax = (bbox.max.x() - ray.origin.x()) * invDir.x();
158
159 if (tmin > tmax) std::swap(tmin, tmax);
160
161 float tymin = (bbox.min.y() - ray.origin.y()) * invDir.y();
162 float tymax = (bbox.max.y() - ray.origin.y()) * invDir.y();
163
164 if (tymin > tymax) std::swap(tymin, tymax);
165
166 if ((tmin > tymax) || (tymin > tmax)) {
167 continue;
168 }
169
170 tmin = std::max(tmin, tymin);
171 tmax = std::min(tmax, tymax);
172
173 float tzmin = (bbox.min.z() - ray.origin.z()) * invDir.z();
174 float tzmax = (bbox.max.z() - ray.origin.z()) * invDir.z();
175
176 if (tzmin > tzmax) std::swap(tzmin, tzmax);
177
178 if ((tmin > tzmax) || (tzmin > tmax)) {
179 continue;
180 }
181
182 tmin = std::max(tmin, tzmin);
183
184 if (tmin > 0.0f && tmin < closestT) {
185 closestT = tmin;
186 closestId = obj->id();
187 }
188 }
189
190 if (closestId != InvalidObjectId) {
191 return closestId;
192 }
193 return std::nullopt;
194}
