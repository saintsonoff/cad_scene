59 /// @brief Create buffer from geometry data.
60 void create(QOpenGLFunctions_4_1_Core* gl, const GeometryData& data) {
61 cleanup();
62
63 m_gl = gl;
64 m_indexCount = static_cast<GLsizei>(data.indices.size());
65
66 if (data.isEmpty()) {
67 return;
68 }
69
70 gl->glGenVertexArrays(1, &m_vao);
71 gl->glBindVertexArray(m_vao);
72
73 gl->glGenBuffers(1, &m_vbo);
74 gl->glBindBuffer(GL_ARRAY_BUFFER, m_vbo);
75 gl->glBufferData(GL_ARRAY_BUFFER,
76 static_cast<GLsizeiptr>(data.vertices.size() * sizeof(Vertex)),
77 data.vertices.data(),
78 GL_STATIC_DRAW);
79
80 gl->glGenBuffers(1, &m_ibo);
81 gl->glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, m_ibo);
82 gl->glBufferData(GL_ELEMENT_ARRAY_BUFFER,
83 static_cast<GLsizeiptr>(data.indices.size() * sizeof(std::uint32_t)),
84 data.indices.data(),
85 GL_STATIC_DRAW);
86
87 // Position (location 0)
88 gl->glEnableVertexAttribArray(0);
89 gl->glVertexAttribPointer(0, 3, GL_FLOAT, GL_FALSE, sizeof(Vertex),
90 reinterpret_cast<void*>(offsetof(Vertex, position)));
91
92 // Normal (location 1)
93 gl->glEnableVertexAttribArray(1);
94 gl->glVertexAttribPointer(1, 3, GL_FLOAT, GL_FALSE, sizeof(Vertex),
95 reinterpret_cast<void*>(offsetof(Vertex, normal)));
96
97 // TexCoord (location 2)
98 gl->glEnableVertexAttribArray(2);
99 gl->glVertexAttribPointer(2, 2, GL_FLOAT, GL_FALSE, sizeof(Vertex),
100 reinterpret_cast<void*>(offsetof(Vertex, texCoord)));
101
102 gl->glBindVertexArray(0);
103 }
