31Result<ObjectId> Application::loadFile(const QString& filePath) {
32 auto result = m_fileLoaderRegistry->load(filePath);
33 if (!result) {
34 emit errorOccurred(result.error().toString());
35 return std::unexpected(result.error());
36 }
37
38 auto geometry = std::make_shared<MeshGeometry>(std::move(result.value()));
39 QFileInfo fileInfo(filePath);
40 auto object = std::make_shared<SceneObject>(geometry, fileInfo.baseName());
41
42 ObjectId id = m_scene->addObject(object);
43
44 emit fileLoaded(filePath, id);
45
46 return id;
47}
